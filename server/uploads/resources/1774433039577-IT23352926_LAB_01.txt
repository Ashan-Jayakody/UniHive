1. CREATE TABLE STRUCTURES

CREATE TABLE client (
    clno CHAR(3) PRIMARY KEY,
    name VARCHAR(12),
    address VARCHAR(30)
);

CREATE TABLE stock (
    company CHAR(7) PRIMARY KEY,
    price REAL,
    dividend REAL,
    eps REAL
);

CREATE TABLE trading (
    company CHAR(7),
    exchange VARCHAR(12),
    FOREIGN KEY(company) REFERENCES stock(company)
);

CREATE TABLE purchase (
    clno CHAR(3),
    company CHAR(7),
    pdate DATE,
    qty INTEGER,
    price REAL,
    FOREIGN KEY(clno) REFERENCES client(clno),
    FOREIGN KEY(company) REFERENCES stock(company)
);

-- 3.QUERIES(Answer)

-- Query (a): Client's name and investment details
SELECT c.name, p.company, s.price, s.dividend, s.eps
FROM client c
JOIN purchase p ON c.clno = p.clno
JOIN stock s ON p.company = s.company;

-- Query (b): Total shares and Average Purchase Price per client
SELECT c.name, p.company, SUM(p.qty) AS Total_Shares, 
       SUM(p.qty * p.price) / SUM(p.qty) AS Avg_Price
FROM client c
JOIN purchase p ON c.clno = p.clno
GROUP BY c.name, p.company;

-- Query (c): New York traded stocks and current value
SELECT p.company, c.name, SUM(p.qty) AS Qty_Held, 
       (SUM(p.qty) * s.price) AS Current_Value
FROM client c
JOIN purchase p ON c.clno = p.clno
JOIN stock s ON p.company = s.company
JOIN trading t ON s.company = t.company
WHERE t.exchange = 'New York'
GROUP BY p.company, c.name;

-- Query (d): Total portfolio cost per client
SELECT c.name, SUM(p.qty * p.price) AS Total_Cost
FROM client c
JOIN purchase p ON c.clno = p.clno
GROUP BY c.name;

-- Query (e): Book Profit or Loss
SELECT c.name, 
       SUM(p.qty * s.price) - SUM(p.qty * p.price) AS Profit_Loss
FROM client c
JOIN purchase p ON c.clno = p.clno
JOIN stock s ON p.company = s.company
GROUP BY c.name;
